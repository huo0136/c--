cmake_minimum_required(VERSION 3.16) 
 
project(birthday_project LANGUAGES CXX) 
 
add_executable(birthday 
    src/main.cpp 
    src/student.cpp 
) 
 
target_compile_features(birthday PRIVATE cxx_std_11) 
target_include_directories(birthday PRIVATE 
"${CMAKE_CURRENT_SOURCE_DIR}/include") 
 
# 便于在 build 目录运行程序时读取输入文件 
configure_file( 
    "${CMAKE_CURRENT_SOURCE_DIR}/students.txt" 
    "${CMAKE_CURRENT_BINARY_DIR}/students.txt" 
    COPYONLY 
)
